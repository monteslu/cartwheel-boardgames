# kilowattcountry (title: Kilowatt Country): bid on power plants, buy fuel, wire up cities on a night map.
#   kilowattcountry              the game
#   kilowattcountry_auto         the self-playing cart (four Normal CPUs, short beats; cart-smoke runs it)
#   kilowattcountry_test_rules   the rules, CPU games and their controls, headless (native only; tests/kilowattcountry-rules.mjs)
#   kilowattcountry_cpu_bench    CPU level against CPU level over seeded games (native only)
set(KC_RULES src/rules.c)
set(KC_SRC src/main.c src/game.c src/view.c ${KC_RULES})
set(KC_META TITLE "Kilowatt Country" DESCRIPTION "Bid on power plants, buy fuel as prices climb and wire up cities across a night map, against friends or the computer." AUTHOR "Luis Montes")
set(KC_ASSETS ASSETS ${CMAKE_CURRENT_SOURCE_DIR}/assets ART ${CMAKE_CURRENT_SOURCE_DIR}/art)
set(KC_PROGRAMS PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/kilowattcountry.txt)
bk_add_app(kilowattcountry SOURCES ${KC_SRC} ${KC_ASSETS} ${KC_PROGRAMS} ${KC_META})
bk_add_app(kilowattcountry_auto SOURCES ${KC_SRC} ${KC_ASSETS} ${KC_PROGRAMS} ${KC_META})
target_compile_definitions(kilowattcountry_auto PRIVATE KILOWATTCOUNTRY_AUTOPILOT=1)
foreach(t kilowattcountry kilowattcountry_auto)
  target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
endforeach()

if(NOT CW_CART)
  add_executable(kilowattcountry_test_rules tests/test_rules.c ${KC_RULES} ${BOARDKIT_RULES_SOURCES})
  add_executable(kilowattcountry_cpu_bench tests/cpu_bench.c ${KC_RULES} ${BOARDKIT_RULES_SOURCES})
  foreach(t kilowattcountry_test_rules kilowattcountry_cpu_bench)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${BOARDKIT_DIR}/..)
    target_link_libraries(${t} PRIVATE cartwheel m)
  endforeach()
endif()
