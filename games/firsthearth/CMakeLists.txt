# firsthearth (title: First Hearth): a village of the first people; dice-driven worker placement.
#   firsthearth              the game
#   firsthearth_auto         the self-playing cart (four Normal CPUs, short beats; cart-smoke runs it)
#   firsthearth_test_rules   the rules, CPU games and their controls, headless (native only; tests/firsthearth-rules.mjs)
#   firsthearth_cpu_bench    CPU level against CPU level over seeded games (native only)
set(FIRSTHEARTH_RULES src/rules.c)
set(FIRSTHEARTH_SRC src/main.c src/game.c src/view.c ${FIRSTHEARTH_RULES})
set(FIRSTHEARTH_META TITLE "First Hearth" DESCRIPTION "Send your kin to hunt, gather and build, roll for every haul and feed the village, against friends or the computer." AUTHOR "Luis Montes")
set(FIRSTHEARTH_ASSETS ASSETS ${CMAKE_CURRENT_SOURCE_DIR}/assets ART ${CMAKE_CURRENT_SOURCE_DIR}/art)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/programs/firsthearth.txt)
  set(FIRSTHEARTH_PROGRAMS PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/firsthearth.txt)
endif()
bk_add_app(firsthearth SOURCES ${FIRSTHEARTH_SRC} ${FIRSTHEARTH_ASSETS} ${FIRSTHEARTH_PROGRAMS} ${FIRSTHEARTH_META})
bk_add_app(firsthearth_auto SOURCES ${FIRSTHEARTH_SRC} ${FIRSTHEARTH_ASSETS} ${FIRSTHEARTH_PROGRAMS} ${FIRSTHEARTH_META})
target_compile_definitions(firsthearth_auto PRIVATE FIRSTHEARTH_AUTOPILOT=1)
foreach(t firsthearth firsthearth_auto)
  target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
endforeach()

if(NOT CW_CART)
  foreach(t firsthearth_test_rules firsthearth_cpu_bench)
    if(t STREQUAL firsthearth_test_rules)
      add_executable(${t} tests/test_rules.c ${FIRSTHEARTH_RULES} ${BOARDKIT_RULES_SOURCES})
    else()
      add_executable(${t} tests/cpu_bench.c ${FIRSTHEARTH_RULES} ${BOARDKIT_RULES_SOURCES})
    endif()
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${BOARDKIT_DIR}/..)
    target_link_libraries(${t} PRIVATE cartwheel m)
  endforeach()
endif()
