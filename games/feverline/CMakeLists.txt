# feverline (title: Fever Line): a cooperative race to cure four fevers across a map of real cities (nine maps).
#   feverline              the game
#   feverline_auto         the self-playing cart (a team of four Normal CPUs on the Normal setup, short beats; cart-smoke runs it)
#   feverline_test_rules   the rules, full games and their controls, headless (native only; tests/feverline-rules.mjs)
#   feverline_cpu_bench    all-CPU teams over seeded games: the win rate per setup and team size (native only)
set(FEVERLINE_RULES src/rules.c)
set(FEVERLINE_SRC src/main.c src/game.c src/view.c ${FEVERLINE_RULES})
set(FEVERLINE_META TITLE "Fever Line" DESCRIPTION "Four fevers spread across a map of real cities, the world or one of eight regions. Travel, treat, share cards and cure them all as one team with every card face up, alone or with friends and computer teammates." AUTHOR "Luis Montes")
set(FEVERLINE_ASSETS ASSETS ${CMAKE_CURRENT_SOURCE_DIR}/assets)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/art/art.json)
  list(APPEND FEVERLINE_ASSETS ART ${CMAKE_CURRENT_SOURCE_DIR}/art)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/programs/feverline.txt)
  set(FEVERLINE_PROGRAMS PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/feverline.txt)
endif()
bk_add_app(feverline SOURCES ${FEVERLINE_SRC} ${FEVERLINE_ASSETS} ${FEVERLINE_PROGRAMS} ${FEVERLINE_META})
bk_add_app(feverline_auto SOURCES ${FEVERLINE_SRC} ${FEVERLINE_ASSETS} ${FEVERLINE_PROGRAMS} ${FEVERLINE_META})
target_compile_definitions(feverline_auto PRIVATE FEVERLINE_AUTOPILOT=1)
foreach(t feverline feverline_auto)
  target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
endforeach()

if(NOT CW_CART)
  foreach(t feverline_test_rules feverline_cpu_bench)
    if(t STREQUAL feverline_test_rules)
      add_executable(${t} tests/test_rules.c ${FEVERLINE_RULES} ${BOARDKIT_RULES_SOURCES})
    else()
      add_executable(${t} tests/cpu_bench.c ${FEVERLINE_RULES} ${BOARDKIT_RULES_SOURCES})
    endif()
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${BOARDKIT_DIR}/..)
    target_link_libraries(${t} PRIVATE cartwheel m)
  endforeach()
endif()
