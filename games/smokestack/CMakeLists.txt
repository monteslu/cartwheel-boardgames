# smokestack (title: Smokestack): build track across a hex valley and haul goods to the cities that want them.
#   smokestack              the game
#   smokestack_auto         the self-playing cart (CPUs on every seat, short beats; cart-smoke runs it)
#   smokestack_test_rules   the rules, CPU games and their controls, headless (native only; tests/smokestack-rules.mjs)
#   smokestack_cpu_bench    CPU level against CPU level over seeded games (native only)
set(SMOKESTACK_RULES src/rules.c)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/main.c)
  set(SMOKESTACK_SRC src/main.c src/game.c src/view.c src/hud.c src/geom.c ${SMOKESTACK_RULES})
  set(SMOKESTACK_META TITLE "Smokestack" DESCRIPTION "Lay track across a river valley, bid for the lead and haul goods to the cities that want them, against friends or the computer." AUTHOR "Luis Montes")
  set(SMOKESTACK_PROGRAMS)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/programs/smokestack.txt)
    set(SMOKESTACK_PROGRAMS PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/smokestack.txt)
  endif()
  bk_add_app(smokestack SOURCES ${SMOKESTACK_SRC} ASSETS ${CMAKE_CURRENT_SOURCE_DIR}/assets ART ${CMAKE_CURRENT_SOURCE_DIR}/art ${SMOKESTACK_PROGRAMS} ${SMOKESTACK_META})
  bk_add_app(smokestack_auto SOURCES ${SMOKESTACK_SRC} ASSETS ${CMAKE_CURRENT_SOURCE_DIR}/assets ${SMOKESTACK_PROGRAMS} ${SMOKESTACK_META})
  target_compile_definitions(smokestack_auto PRIVATE SMOKESTACK_AUTO=1)
  foreach(t smokestack smokestack_auto)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  endforeach()
endif()

if(NOT CW_CART)
  add_executable(smokestack_test_rules tests/test_rules.c ${SMOKESTACK_RULES} ${BOARDKIT_RULES_SOURCES})
  add_executable(smokestack_cpu_bench tests/cpu_bench.c ${SMOKESTACK_RULES} ${BOARDKIT_RULES_SOURCES})
  foreach(t smokestack_test_rules smokestack_cpu_bench)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${BOARDKIT_DIR}/..)
    target_link_libraries(${t} PRIVATE cartwheel m)
  endforeach()
endif()
